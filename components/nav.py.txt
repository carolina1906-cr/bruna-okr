    with cols[0]:

    with cols[1]:
        st.page_link("pages/2_Registrar_datos.py", label="📋 Registrar datos", use_container_width=True)

    with cols[3]:
        st.page_link("pages/4_Exportar.py", label="📥 Exportar", use_container_width=True)
    st.divider()
